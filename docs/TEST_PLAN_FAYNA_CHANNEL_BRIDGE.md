## 0. Як запускати (гейт, якого зараз немає)

CI має запускати **реальні Odoo-тести**, а не лише lint. Мінімальний test-job:

```yaml

    name: Odoo tests (coverage gate)
    runs-on: ubuntu-latest
    container: odoo:17.0          # той самий образ, що на проді — не pip install odoo
    services:
      postgres:
        image: postgres:15
        env: { POSTGRES_USER: odoo, POSTGRES_PASSWORD: odoo, POSTGRES_DB: postgres }
        options: >-
          --health-cmd "pg_isready -U odoo"
          --health-interval 10s --health-timeout 5s --health-retries 5
    steps:
      - uses: actions/checkout@v4
      - name: Deps
        run: pip install --break-system-packages coverage requests
      - name: Run module tests
        run: |
          coverage run --source=addons/fayna_channel_bridge \
            /usr/bin/odoo -d test_db --db_host postgres -r odoo -w odoo \
            --addons-path=/usr/lib/python3/dist-packages/odoo/addons,addons \
```

## 1. Критичний наскрізний сценарій оператора (ловить F-01, F-02, F-04, F-05 разом)

Це **найважливіший тест** — він відтворює реальний шлях «вебхук → розмова →
оператор бачить → оператор відповідає → клієнт отримує». Саме цього шляху не
покриває жоден наявний тест.

### T-01 · Вебхук → розмова → оператор у каналі → відповідь → доставка

> 🔴 **Тест мусить іти через HTTP-контролер, а не кликати
> `_process_incoming_event` напряму.** Другий `create` (корінь F-04) живе саме
> в парі «контролер + метод моделі». Виклик лише методу **фізично не може**
> відтворити дубль — такий тест був би декорацією.

```python
import json
from unittest.mock import Mock, patch

from odoo.tests import HttpCase, tagged


@tagged('post_install', '-at_install')
class TestOperatorEndToEnd(HttpCase):
    def setUp(self):
        super().setUp()
        self.backend = self.env['channel.backend'].create({
            'name': 'TG', 'service': 'telegram', 'provider': 'direct',
            'transport_priority': 'own', 'bot_token': '123:TOK', 'bot_id': '@Bot',
            'user_ids': [(6, 0, [self.env.ref('base.user_admin').id])],
        })
        self.officer = self.env.ref('fayna_channel_bridge.group_channel_bridge_officer')

    @staticmethod
    def _tg_ok(status=200, message_id=777):
        """Мок відповіді Telegram Bot API (був не визначений — дефект плану v1)."""
        resp = Mock()
        resp.status_code = status
        resp.text = '{"ok": true}'
        resp.json.return_value = {'ok': True, 'result': {'message_id': message_id}}
        return resp

    def _incoming(self, text, mid=100, uid=12345):
        """Справжній POST на webhook-роут — єдиний шлях, що відтворює F-04."""
        payload = {'update_id': mid, 'message': {
            'message_id': mid, 'date': 1756500000,
            'chat': {'id': uid}, 'from': {'first_name': 'Клієнт'}, 'text': text}}
        r = self.url_open('/bridge/telegram/webhook/123:TOK',
                          data=json.dumps(payload),
                          headers={'Content-Type': 'application/json'})
        self.assertEqual(r.status_code, 200, 'webhook має відповісти 200')
        return self.env['channel.conversation'].search(
            [('provider_user_id', '=', str(uid))], limit=1)

    def test_full_operator_flow(self):
        conv = self._incoming('Потрібна допомога')
        self.assertTrue(conv, 'розмову має бути створено вебхуком')
        # F-01: канал створено і оператор є учасником
        self.assertTrue(conv.channel_id)
        members = self.env['discuss.channel.member'].search_count(
            [('channel_id', '=', conv.channel_id.id)])
        self.assertGreater(members, 0, 'F-01: оператор має бути учасником каналу')
        # F-04: рівно 1 рядок у журналі
        n = self.env['channel.message'].search_count(
            [('provider_message_id', '=', '100'), ('direction', '=', 'incoming')])
        self.assertEqual(n, 1, 'F-04: один webhook = один рядок журналу')
        # F-02: відповідь оператора не падає і йде клієнту
        with patch('requests.post') as mock:
            mock.return_value = self._tg_ok(200)
            conv.channel_id.with_user(self.env.ref('base.user_admin')).message_post(
                body='<p>Відповідь</p>', message_type='comment',
                subtype_xmlid='mail.mt_comment')
        # message_post не кинув AttributeError (F-02)
        sent = self.env['channel.message'].search(
            [('direction', '=', 'outgoing'), ('text_message', '=', 'Відповідь')])
        self.assertEqual(len(sent), 1, 'F-02: рівно одне вихідне')
        self.assertEqual(sent.state, 'sent')
        # транспорт реально викликано
        self.assertTrue(mock.called, 'F-02: транспорт має отримати виклик')
```

**Критерій приймання:** тест **падає** на поточному коді (F-01: 0 учасників;
F-02: `AttributeError`; F-04: 2 рядки), **проходить** після фіксів Етапу 1.
